// ---- PvP kill counter (shown in the website's Hall of Fame) ----
// Every time a player kills another player, the killer's PVP_KILLS goes up by one
// (a permanent character variable, stored in the char_reg_num table).
-	script	DudsPvPCounter	-1,{
OnPCKillEvent:
	PVP_KILLS++;
	dispbottom "PvP: you slayed " + rid2name(killedrid) + "! (" + PVP_KILLS + " kills)";
	end;
}
