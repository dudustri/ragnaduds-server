// inside the Tourney Hall (instance on 1@4igd): the ring outside, the arena inside (only reachable by warp)
1@4igd,50,30,4	script	King's Knight#d4igd4	20823,{
	.@n$ = "[King's Knight]";
	.@m$ = instance_mapname("1@4igd");
	mes .@n$;
	if ('igd_state >= 3) {
		mes "You won. Come, step into the hall and receive your crest.";
		next;
		if (select("Enter the hall:Leave the Tourney Hall") == 2) { close2; warp "prt_cas", 181, 13; end; }
		close2;
		warp .@m$, 50, 42;
		end;
	}
	if ('igd_state == 0) {
		mes "Welcome to the Tourney Hall of Trials.";
		mes "Inside, the ^0000FFKnight of Devotion^000000 (4.8M HP) will face you first, then the ^0000FFKnight of Armament^000000 (5M HP).";
		mes "Defeat both and the crest of the Imperial Guard is yours.";
	} else
		mes "The trial is still going on. The knights are waiting for you in the hall.";
	next;
	if (select("Enter the hall:Leave the Tourney Hall") == 2) { close2; warp "prt_cas", 181, 13; end; }
	if ('igd_state == 0)
		'igd_state = 1;
	// (re)spawn the current knight if it's not there
	if ('igd_state == 1 && mobcount(.@m$, instance_npcname("King's Knight#d4igd4") + "::OnDevotion") == 0)
		monster .@m$, 50, 54, "Knight of Devotion", 20825, 1, instance_npcname("King's Knight#d4igd4") + "::OnDevotion";
	if ('igd_state == 2 && mobcount(.@m$, instance_npcname("King's Knight#d4igd4") + "::OnArmament") == 0)
		monster .@m$, 50, 54, "Knight of Armament", 20826, 1, instance_npcname("King's Knight#d4igd4") + "::OnArmament";
	close2;
	warp .@m$, 50, 42;
	mapannounce .@m$, ('igd_state == 1 ? "Knight of Devotion: Show me your faith, Royal Guard!" : "Knight of Armament: Now show me your loyalty!"), bc_map, "0xFFD700";
	end;

OnDevotion:
	if ('igd_state != 1) end;
	'igd_state = 2;
	.@m$ = instance_mapname("1@4igd");
	mapannounce .@m$, "Knight of Devotion: Your faith is real... The Knight of Armament steps into the hall!", bc_map, "0xFFD700";
	sleep 3000;
	if ('igd_state == 2 && mobcount(.@m$, instance_npcname("King's Knight#d4igd4") + "::OnArmament") == 0)
		monster .@m$, 50, 54, "Knight of Armament", 20826, 1, instance_npcname("King's Knight#d4igd4") + "::OnArmament";
	end;

OnArmament:
	if ('igd_state != 2) end;
	'igd_state = 3;
	.@m$ = instance_mapname("1@4igd");
	enablenpc instance_npcname("King's Knight#d4igd5");
	mapannounce .@m$, "Both knights have fallen! The King's Knight enters the hall. Talk to him.", bc_map, "0xFFD700";
	end;
}
