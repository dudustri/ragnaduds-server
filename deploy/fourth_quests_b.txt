//= guides: steps, NPC names, places and trials). rAthena ships none of them; dialogue is ours.
//= Shared helpers (F_Duds4th_Is / _Levels / _Promote / _Enter) live in fourth_quests.txt.
//= Private instances (re_db_import/instance_db.yml):
//=   923 "Arch Mage Trial"        1@4mag (official Arch Mage map)
//=   924 "Elemental Master Trial" 1@4mag (same Varmundt map: right room, north hall, south exit)
//=   925 "Troubadour Trial"       1@4tro (official Troubadour / Trouvere map)
//= Both 1@4mag quests share the map: every instance NPC checks the instance name and hides itself in the other
//= quest. The public copy of 1@4mag is used as Yoop's hideout (isolated room in the south-west, 66-78 / 37-47).
//= Requirements: base 200 / job 70. Reward: the job change + Hourglass Necklace (490087, in F_Duds4th_Promote).
//= Progress variables:
//=   DUDS4_AMG   Arch Mage stage (0-7), DUDS4_AMG_K hunt counter, DUDS4_AMG_I bit mask of items already used
//=               inside the trial (1 Burning Heart, 2 Mermaid's Heart, 4 Yggdrasil Seed: never lost on a retry)
//=   DUDS4_ELM   Elemental Master stage (0-9), DUDS4_ELM_A / DUDS4_ELM_B hunt counters
//=   DUDS4_TRO   Troubadour / Trouvere stage (0-4), DUDS4_TRO_A monster task (0-7), DUDS4_TRO_B dialogue
//=               task (0-4), DUDS4_TRO_K hunt counter

// getarg(0): monster GID, getarg(1): hp. Sets the max and current HP of a spawned monster.
function	script	F_Duds4thB_Hp	{
	setunitdata getarg(0), UMOB_MAXHP, getarg(1);
	setunitdata getarg(0), UMOB_HP, getarg(1);
	return;
}

// hunting counters of the three quests
-	script	Duds4thKillsB	-1,{
OnNPCKillEvent:
	.@map$ = strcharinfo(3);
	// Arch Mage: the three hearts / seed Maggie asks for
	if (DUDS4_AMG >= 3 && DUDS4_AMG <= 6) {
		if (.@map$ == "thor_v03" && !countitem(7097) && !(DUDS4_AMG_I & 1)) {
			DUDS4_AMG_K++;
			if (rand(100) < 10 || DUDS4_AMG_K >= 20) {
				DUDS4_AMG_K = 0;
				getitem 7097, 1;   // Burning Heart
				dispbottom "Maggie's task: a Burning Heart fell from the volcano monster! Bring it to Maggie.";
			}
		} else if (.@map$ == "ba_bath" && DUDS4_AMG >= 4 && !countitem(950) && !(DUDS4_AMG_I & 2)) {
			DUDS4_AMG_K++;
			if (rand(100) < 10 || DUDS4_AMG_K >= 20) {
				DUDS4_AMG_K = 0;
				getitem 950, 1;    // Mermaid's Heart
				dispbottom "Maggie's task: you found a Mermaid's Heart in the bath water! Bring it to Maggie.";
			}
		} else if (.@map$ == "odin_tem03" && DUDS4_AMG >= 5 && killedrid == 1083 && !countitem(608) && !(DUDS4_AMG_I & 4)) {
			getitem 608, 1;        // Yggdrasil Seed
			dispbottom "Maggie's task: the Shining Plant left a Yggdrasil Seed! Bring it to Maggie.";
		}
	}
	// Elemental Master: Power Plant 1 and the library
	if (DUDS4_ELM == 5 && .@map$ == "ba_pw01") {
		if (killedrid == 20689 && DUDS4_ELM_A < 10) {
			DUDS4_ELM_A++;
			dispbottom "Potent Mana: Spell Addicted Dolor " + DUDS4_ELM_A + " / 10, Released Spell " + DUDS4_ELM_B + " / 10";
		} else if (killedrid == 20690 && DUDS4_ELM_B < 10) {
			DUDS4_ELM_B++;
			dispbottom "Potent Mana: Spell Addicted Dolor " + DUDS4_ELM_A + " / 10, Released Spell " + DUDS4_ELM_B + " / 10";
		}
		if (DUDS4_ELM_A >= 10 && DUDS4_ELM_B >= 10)
			dispbottom "The mana calms down. Touch the Potent Mana Stone again (centre of the Power Plant).";
	} else if (DUDS4_ELM == 7 && .@map$ == "ba_lib" && killedrid == 20684 && DUDS4_ELM_A < 10) {
		DUDS4_ELM_A++;
		dispbottom "Potent Mana: Roaming Spellbook " + DUDS4_ELM_A + " / 10" + (DUDS4_ELM_A >= 10 ? ". Go back to the Potent Mana Bookshelf." : "");
	}
	// Troubadour / Trouvere: the three Small Pieces
	if (DUDS4_TRO_A == 2 && .@map$ == "niflheim" && DUDS4_TRO_K < 20 && getmonsterinfo(killedrid, MOB_LV) >= 105) {
		DUDS4_TRO_K++;
		dispbottom "Small Piece (Niflheim): " + DUDS4_TRO_K + " / 20 monsters of level 105+" + (DUDS4_TRO_K >= 20 ? ". Return to the Small Piece." : "");
	} else if (DUDS4_TRO_A == 4 && .@map$ == "thor_v01" && DUDS4_TRO_K < 15 && getmonsterinfo(killedrid, MOB_ELEMENT) == ELE_FIRE && getmonsterinfo(killedrid, MOB_LV) >= 130) {
		DUDS4_TRO_K++;
		dispbottom "Small Piece (Thor Volcano): " + DUDS4_TRO_K + " / 15 fire monsters of level 130+" + (DUDS4_TRO_K >= 15 ? ". Return to the Small Piece." : "");
	} else if (DUDS4_TRO_A == 6 && .@map$ == "abyss_03" && DUDS4_TRO_K < 10 && getmonsterinfo(killedrid, MOB_RACE) == RC_DRAGON && getmonsterinfo(killedrid, MOB_LV) >= 120) {
		DUDS4_TRO_K++;
		dispbottom "Small Piece (Abyss Lake): " + DUDS4_TRO_K + " / 10 dragons of level 120+" + (DUDS4_TRO_K >= 10 ? ". Return to the Small Piece." : "");
	}
	end;
}
