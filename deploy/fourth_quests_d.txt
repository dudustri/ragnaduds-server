// shows the cabin gauges (args: day, food, water, fire lit, points, animals)
function	script	F_D4Whk_Show	{
	dispbottom "Day " + getarg(0) + "/10 | Food " + getarg(1) + "% | Water " + getarg(2) + "% | Fireplace " + (getarg(3) ? "lit" : "OUT") + " | Animals " + getarg(5) + " | Survival points " + getarg(4), 0xFFDD66;
	return;
}

// tunes a quest boss: getarg(0) GID, getarg(1) HP. Normal damage taken (the db has 10%), so HP is the real HP.
function	script	F_D4_Boss	{
	setunitdata getarg(0), UMOB_MAXHP, getarg(1);
	setunitdata getarg(0), UMOB_HP, getarg(1);
	setunitdata getarg(0), UMOB_DAMAGETAKEN, 100;
	return;
}

1@4win,56,150,4	script	Survival Diary#d4whk	4_BULLETIN_BOARD2,{
	mes "^777777Survival Diary^000000";
	if ('whk_st < 3) {
		mes "\"Whoever finds this: the cabin key is hidden in the ^0000FFflowers by the door^000000.\"";
	} else if ('whk_st == 3) {
		mes "Day ^0000FF" + 'whk_day + " / 10^000000 - Survival points ^0000FF" + 'whk_pts + "^000000 (below 50 and you're done for)";
		mes "Food Storage " + 'whk_food + "% - Drinking Water " + 'whk_water + "% - Fireplace " + ('whk_fire ? "lit" : "^FF0000out^000000") + " - Animals " + 'whk_pets;
		mes "From ^0000FFday 3^000000, every night: Fireplace lit, Food above 40%, Water above 30%, animals fed and warm.";
	} else {
		mes "\"Ten days. You made it. The owner is waiting by the collapsed rocks.\"";
	}
	next;
	switch (select("How do I survive here?:Close:Leave the forest")) {
	case 1:
		mes "^777777Survival Diary^000000";
		mes "- ^0000FFWarehouse^000000 (left of the cabin): take the hoe, the bucket or the axe. One tool at a time.";
		mes "- ^0000FFFirewood Pile^000000 (2 o'clock, needs the axe): chop wood, feed the ^0000FFFireplace^000000 (2 bundles keep it lit all night).";
		mes "- ^0000FFSpring^000000 (1 o'clock, needs the bucket): fill it and pour it in the Drinking Water Bucket: +10% each.";
		mes "- ^0000FFRed plants^000000 around the clearing (need the hoe): +15% food each, they grow back every day.";
		mes "- ^0000FFTraps^000000 (11 o'clock): set two every day. Next morning: make food (+80%) or keep the animal in the basket (feed it 10% food daily, keep the fire lit).";
		mes "- Sleep in the ^0000FFbed^000000 (Food above 40%, Water above 30%) to start the next day. A day lasts at most 10 minutes.";
		close;
	case 2:
		close;
	case 3:
		mes "Leave the forest? The cabin waits while this place lasts; come back through the frog statue.";
		next;
		if (select("Leave:Stay") == 2) close;
		close2;
		warp "um_fild01", 47, 341;
		end;
	}
	end;

OnStartDay:   // a new day: overnight consumption, traps, timer
	.@m$ = instance_mapname("1@4win");
	'whk_fire = 0;
	'whk_fuel = 0;
	'whk_plant = 0;
	'whk_fed = 0;
	'whk_slept = 0;
	if ('whk_day > 1) {
		'whk_food = max(0, 'whk_food - 15);
		'whk_water = max(0, 'whk_water - 20);
		for (.@i = 1; .@i <= 2; .@i++) {
			if ('whk_trap[.@i] == 1)
				'whk_trap[.@i] = (rand(100) < 25) ? 4 : 2;   // 4 = empty this morning, 2 = caught something
		}
	}
	mapannounce .@m$, "Day " + 'whk_day + " / 10 begins. Food " + 'whk_food + "%, Water " + 'whk_water + "%, the Fireplace is out. Survival points: " + 'whk_pts + ".", bc_map, "0xFFDD66";
	if ('whk_day == 3)
		mapannounce .@m$, "From today on, every night counts: Fireplace lit, Food above 40%, Water above 30%, animals fed.", bc_map, "0xFF9966";
	initnpctimer;
	end;

OnTimer540000:
	mapannounce instance_mapname("1@4win"), "The sun is setting. One minute left before you collapse from exhaustion: go to bed!", bc_map, "0xFF9966";
	end;

OnTimer600000:   // the day ran out: collapse (no sleep)
	'whk_slept = 0;
	// fall through
OnEndDay:
	stopnpctimer;
	if ('whk_st != 3) end;
	.@m$ = instance_mapname("1@4win");
	.@lost = 0;
	.@why$ = "";
	if ('whk_day >= 3) {
		if (!'whk_slept) { .@lost += 10; .@why$ += " exhausted -10"; }
		if (!'whk_fire) { .@lost += 10; .@why$ += " cold fireplace -10"; }
		if ('whk_food <= 40) { .@lost += 10; .@why$ += " hungry -10"; }
		if ('whk_water <= 30) { .@lost += 10; .@why$ += " thirsty -10"; }
		if ('whk_pets && !'whk_fed) { .@lost += 5 * 'whk_pets; .@why$ += " animals unfed -" + (5 * 'whk_pets); }
		if ('whk_pets && !'whk_fire) { .@lost += 5 * 'whk_pets; .@why$ += " animals cold -" + (5 * 'whk_pets); }
	}
	'whk_pts -= .@lost;
	if (.@lost)
		mapannounce .@m$, "Night " + 'whk_day + ":" + .@why$ + ". Survival points: " + 'whk_pts + ".", bc_map, "0xFF7777";
	else
		mapannounce .@m$, "Night " + 'whk_day + ": a peaceful night. Survival points: " + 'whk_pts + ".", bc_map, "0x77FF77";
	if ('whk_pts < 50) {
		'whk_st = 9;
		mapannounce .@m$, "You couldn't survive the forest... You are carried out. Turn the frog statue east to try again.", bc_map, "0xFF7777";
		sleep 4000;
		instance_warpall "um_fild01", 47, 341, instance_id();
		instance_destroy;
		end;
	}
	if ('whk_day >= 10) {
		'whk_st = 4;
		enablenpc instance_npcname("Silla#d4whk");
		mapannounce .@m$, "Morning of the 11th day: voices behind the collapsed rocks! Someone cleared the path. Go to the rocks.", bc_map, "0xFFD700";
		end;
	}
	'whk_day++;
	donpcevent instance_npcname("Survival Diary#d4whk") + "::OnStartDay";
	end;
}
