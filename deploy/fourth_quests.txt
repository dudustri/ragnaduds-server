//===== RagnaDuds: 4th job change quests (renewal) ============================
//= Cardinal, Biolo and Dragon Knight, following the official quests (iRO Wiki job change guides: steps, NPC
//= names, places and trials). rAthena ships none of them; dialogue is ours. The Job Master in Prontera keeps
//= its direct change. Each quest uses a private instance (re_db_import/instance_db.yml, Ids 916-918) on the
//= official 4th job maps the 2026 client has (1@4cdn, 1@4drk) or a private copy of ver_eju (the lab).
//= Requirements: base 200 / job 70 (like the Job Master). Monsters: official 4th job quest Ids (re_db_import/mob_db.yml).
//= Reward, like the official quests: the job change + Hourglass Necklace (490087).
//= Progress: DUDS4_CDN / DUDS4_CDN_K (Cardinal), DUDS4_DK (Dragon Knight), DUDS4_BIO / DUDS4_BIO_B (Biolo).

// getarg(0): the 4th job. True if the player is a (non-baby) 3rd class whose 4th class is that job.
function	script	F_Duds4th_Is	{
	if (!(eaclass()&EAJL_THIRD) || (eaclass()&(EAJL_FOURTH|EAJL_BABY)))
		return false;
	return roclass(eaclass()|EAJL_FOURTH) == getarg(0);
}

// getarg(0): npc title, getarg(1): job level needed. Ends the dialogue if the levels are too low.
function	script	F_Duds4th_Levels	{
	if (BaseLevel >= 200 && JobLevel >= getarg(1))
		return;
	mes getarg(0);
	mes "You have the calling, but not the strength yet.";
	mes "Come back at ^0000FFbase level 200^000000 and ^0000FFjob level " + getarg(1) + "^000000.";
	close;
}

// getarg(0): the 4th job. Same safety checks as the Job Master, then the change and the reward.
function	script	F_Duds4th_Promote	{
	if (checkfalcon() || checkcart() || checkriding() || ismounting()) {
		mes "Remove your cart, falcon or mount first (equipment window), then talk to me again.";
		close;
	}
	if (SkillPoint) {
		mes "Use all your skill points first, or they'd be lost with the job change.";
		close;
	}
	jobchange getarg(0);
	getitem 490087, 1;   // Hourglass Necklace
	specialeffect2 EF_ANGEL2;
	announce strcharinfo(0) + " has become a " + jobname(getarg(0)) + "!", bc_all, "0xFFD700";
	return;
}

// getarg(0): instance name, getarg(1): x, getarg(2): y. Creates (or reuses) the player's private instance.
function	script	F_Duds4th_Enter	{
	.@inst = instance_id(IM_CHAR);
	if (.@inst > 0 && instance_live_info(ILI_NAME, .@inst) != getarg(0)) {
		instance_destroy .@inst;   // one private instance per character (e.g. a Homunculus Room left open)
		.@inst = 0;
	}
	if (.@inst <= 0)
		.@inst = instance_create(getarg(0), IM_CHAR);
	if (.@inst <= 0 || instance_enter(getarg(0), getarg(1), getarg(2), getcharid(0), .@inst) != IE_OK) {
		dispbottom "The place is not available right now. Try again in a moment, or tell Duds.";
		end;
	}
	return;
}

	setunitdata 'dk_gold, UMOB_ATKMIN, 1;
	setunitdata 'dk_gold, UMOB_ATKMAX, 1;
	setunitdata 'dk_gold, UMOB_MATKMIN, 1;
	setunitdata 'dk_gold, UMOB_MATKMAX, 1;
	mapannounce .@m$, "A Golden Acidus appeared! Lure it to Fountain of Life " + 'dk_f + ".", bc_map, "0xFFD700";

	end;

	stopnpctimer;
	if ('dk_stage != 2 || 'dk_fail) end;
	setarray .@fx[1], 60, 82, 102, 112, 84, 132, 142, 127, 155, 72;
	setarray .@fy[1], 50, 36, 35, 45, 52, 44, 58, 56, 51, 54;
	if (getunitdata('dk_gold, .@d) == -1) end;
	.@dx = .@d[UMOB_X] - .@fx['dk_f];
	.@dy = .@d[UMOB_Y] - .@fy['dk_f];
	if (.@dx * .@dx + .@dy * .@dy > 9) {
		initnpctimer;
		end;
	}
	.@m$ = instance_mapname("1@4drk");
	killmonster .@m$, instance_npcname("Oscar#dk") + "::OnGolden";   // no death event: not a failure
	specialeffect EF_HEAL2, AREA, instance_npcname("Fountain of Life#" + 'dk_f);
	if ('dk_f >= 10) {
		disablenpc instance_npcname("Fountain of Life#10");
		'dk_stage = 3;
		enablenpc instance_npcname("Letizia#dk");
		mapannounce .@m$, "The Fountains of Life are restored! Letizia awaits at the top of the path (north-east).", bc_map, "0xFFD700";
		end;
	}
	'dk_f++;
	donpcevent instance_npcname("Oscar#dk") + "::OnFountain";
	end;

OnGolden:   // the Golden Acidus was killed: trial failed
	if ('dk_stage != 2) end;
	'dk_fail = 1;
	stopnpctimer;
	killmonster instance_mapname("1@4drk"), instance_npcname("Oscar#dk") + "::OnBone";
	mapannounce instance_mapname("1@4drk"), "The Golden Acidus died! Talk to Oscar to try the Trial of the Protector again.", bc_map, "0xFF7777";
	end;
